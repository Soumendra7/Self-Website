import java.util.*;

class Black_Hole_Word
{
    public static void Black_Hole_Word(String str , String original_str)  //creating a recursion until we get 4
    {

        String a[] = {"one" , "two" , "three" , "four" , "five" , "six" , "seven" , "eight" , "nine" , "ten" , "eleven" , "twelve" , "thirteen" , "fourteen" , "fifteen" , "sixteen" , "seventeen" , "eighteen" , "nineteen" , "twenty"}; //creating a array to store numbers in their letter form upto 20

        if(str.length() == 4)  //base case
        {
        System.out.println(str+" contains "+str.length()+" letters.");
        System.out.println("Therefore, "+original_str+" is a BLACK HOLE WORD.");
        return;
        }
       
        System.out.println(str+" contains "+str.length()+" letters."); 
        str = a[str.length()-1]; //to get the word form of a number from the declared array
        Black_Hole_Word(str,original_str); //recursion

    }

    public static void main(String[] args) //declaring main function
    {
        Scanner ob = new Scanner(System.in);  //creating Scanner class object
        System.out.print("Enter a word -> ");  //displaying message to user
        String str,original_str;
        str = ob.next();  //taking a word as input from the user
        original_str = str;  //storing the original word entered by the user
        Black_Hole_Word(str,original_str);  //calling recursion named -> Black_Hole_Word
    }
}
