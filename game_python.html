import random # using random class to include random method

lost=0
won=0
print("Hello coder")
ans=input("Do you want to play :) ")

if(ans.lower()=="yes"):
    user_ch=input("Enter \"rock\", \"paper\", \"scissor\" or \"q\" to quit :) ") # initially taking input to enter loop
   
    while(user_ch!='q'):
        
        
        cmp_ch=random.randrange(0,3) #using random method to generate computer choice
        weapon=["rock","paper","scissor"] #declaring all weapons in a list
        
        
        #if user choose rock
        if(user_ch.lower()=="rock"):
            
            if(cmp_ch==0): #rock
                print("it's a tie")
                print("Computer choose",weapon[cmp_ch])
           
            if(cmp_ch==1): #paper
                print("You Loose")
                print("Computer choose",weapon[cmp_ch])
                lost+=1
           
            if(cmp_ch==2): #scissor
                print("You Won")
                print("Computer choose",weapon[cmp_ch])
                won+=1
        

        #if user choose paper
        if(user_ch.lower()=="paper"):

            if(cmp_ch==0): #rock
                print("You Won")
                print("Computer choose",weapon[cmp_ch])
                won+=1

            if(cmp_ch==1): #paper
                print("it's a tie")
                print("Computer choose",weapon[cmp_ch])

            if(cmp_ch==2): #scissor
                print("You Loose")
                print("Computer choose",weapon[cmp_ch])
                lost+=1


        #if user choose scissor
        if(user_ch.lower()=="scissor"):

            if(cmp_ch==0): #rock
                print("You Loose")
                print("Computer choose",weapon[cmp_ch])
                lost+=1

            if(cmp_ch==1): #paper  
                print("You Won")
                print("Computer choose",weapon[cmp_ch])
                won+=1

            if(cmp_ch==2): #scissor
                print("it's a tie")
                print("Computer choose",weapon[cmp_ch])


        user_ch=input("Enter \"rock\", \"paper\", \"scissor\" or \"q\" to quit :) ") # again to take input


print("You won",won,"times")
print("You won",lost,"times")


if(won>lost):
    print("You won by",(won-lost),"points")
if(lost>won):
    print("You losy by",(lost-won),"points")
if(won>0 and lost>0 and won==lost):
    print("It's a tie")
