GST

Description: Replica of my 10th GST project.

print("WELCOME TO TECH WORLD")
name=input("Enter your name coder:-")
ch=input("Enter \"Intra\" for Intra State Transaction \nEnter \"Inter\" for Inter State Transaction \nEnter your choice:-")
print("\n\n\n")


if(ch=="Intra" or ch=="intra"):

    price=float(input("Enter price (IN \u20B9):-")) #to take price
    dis=float(input("Enter discount (IN %):-")) #to take discount
    gst=float(input("Enter GST (IN %):-")) #to take gst
    print(" ")
    print("Here is your bill "+name)
    print(" ")
    print("Price = \u20B9",price) #to print entered price
    print("Discount =",dis,"%") #to print entered discount
    print("GST =",gst,"%") #to print entered gst
    dis_value=price-(price*(dis/100)) #to calculate discounted value
    print("Discount = \u20B9",price*(dis/100)) #to calculate and print discount
    print("Discounted value = \u20B9",dis_value) #to show discounted value
    gst_value=dis_value*(gst/100) #to calculate total GST
    print("CGST applicable = \u20B9",(gst_value/2)) #to show CGST on discounted value
    print("SGST applicable = \u20B9",(gst_value/2)) #to show SGST on discounted value
    print("Total GST = \u20B9",gst_value) #to show total GST
    bill=dis_value+gst_value #to calculate bill
    print("Total bill = \u20B9",bill) #to print bill

elif(ch=="Inter" or ch=="inter"):

    price=float(input("Enter price (IN \u20B9):-")) #to take price
    dis=float(input("Enter discount (IN %):-")) #to take discount
    gst=float(input("Enter GST (IN %):-")) #to take gst
    print(" ")
    print("Here is your bill",name)
    print(" ")
    print("Price = \u20B9",price) #to print entered price
    print("Discount =",dis,"%") #to print entered discount
    print("GST =",gst,"%") #to print entered gst
    dis_value=price-(price*(dis/100)) #to calculate discounted value
    print("Discount = \u20B9",price*(dis/100)) #to calculate and print discount
    print("Discounted value = \u20B9",dis_value) #to show discounted value
    gst_value=dis_value*(gst/100) #to calculate total GST
    print("IGST applicable = \u20B9",gst_value) #to show IGST on discounted value
    bill=dis_value+gst_value #to calculate bill
    print("Total bill = \u20B9",bill) #to print bill

else:

    print("Invalid choice",name,"!")

print("Have a great day :)")
